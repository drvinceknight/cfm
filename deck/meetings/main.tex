\slidehead{What the project actually is}

\vspace{1.3cm}

\begin{center}
{\LARGE You pick a \textbf{problem in mathematics}.\\[0.7em]
 You write a small \textbf{Python library} that does something
 useful with it.\\[0.7em]
 You write a \textbf{three page paper} about it, and you
 \textbf{record a presentation}.}
\end{center}

\vspace{1.4cm}

\begin{center}
{\large\color{inkgrey} That is the whole project. Everything else on the
website is detail about how it is marked.}
\end{center}
